    - computeSkipFilterStats({ Checking: '__SKIP__', Savings: 'acct-1' }, { Checking: 10, Savings: 5 }) returns skippedAccountNames Set(['Checking']), skippedRowCount 10
    - computeSkipFilterStats({ A: '__SKIP__', B: '__SKIP__' }, { A: 3, B: 7 }) returns skippedAccountNames Set(['A','B']), skippedRowCount 10
    - computeSkipFilterStats({ A: '__SKIP__' }, {}) returns skippedAccountNames Set(['A']), skippedRowCount 0 (missing rowCount gracefully handled)
    - computeSkipFilterStats({ A: '', B: 'acct-1' }, { A: 5, B: 10 }) returns skippedAccountNames empty set (empty string is undecided, not skipped), skippedRowCount 0
  </behavior>
  <action>
    Add a new exported pure function `computeSkipFilterStats` to ImportPage.tsx, placed right after the existing `getValidationState` function (around line 40):

    ```typescript
    export function computeSkipFilterStats(
      accountMappings: Record<string, string>,
      rowCountByAccount: Record<string, number>
    ): { skippedAccountNames: Set<string>; skippedRowCount: number } {
      const skippedAccountNames = new Set(
        Object.entries(accountMappings)
          .filter(([, v]) => v === SKIP_SENTINEL)
          .map(([k]) => k)
      );
      const skippedRowCount = [...skippedAccountNames]
        .reduce((sum, name) => sum + (rowCountByAccount[name] ?? 0), 0);
      return { skippedAccountNames, skippedRowCount };
    }
    ```

    Then add tests for this function in ImportPage.test.ts, importing `computeSkipFilterStats` alongside the existing imports. Follow the existing test style (describe block with multiple `it` cases).
